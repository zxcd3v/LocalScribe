cmake_minimum_required(VERSION 3.22)
project(localscribe_llama LANGUAGES C CXX)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(LOCALSCRIBE_AVX_VARIANT "Optional runtime-selected x86 AVX/F16C backend" OFF)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_KLEIDIAI OFF CACHE BOOL "" FORCE)
set(GGML_CCACHE OFF CACHE BOOL "" FORCE)
set(GGML_CPU_REPACK OFF CACHE BOOL "" FORCE)
if(ANDROID AND ANDROID_ABI STREQUAL "x86_64")
    # SSE4.2 is guaranteed by Android's x86_64 ABI; GGML defaults disable it.
    set(GGML_SSE42 ON CACHE BOOL "" FORCE)
    # Android's x86_64 ABI does not guarantee AVX2/FMA. Some AVDs lack them.
    foreach(extension AVX AVX2 FMA F16C BMI2)
        set(GGML_${extension} OFF CACHE BOOL "" FORCE)
    endforeach()
    if(LOCALSCRIBE_AVX_VARIANT)
        set(GGML_AVX ON CACHE BOOL "" FORCE)
        set(GGML_F16C ON CACHE BOOL "" FORCE)
    endif()
endif()
add_subdirectory("${LLAMA_SOURCE_DIR}" llama)
add_library(localscribe_llama SHARED editor.cpp)
target_compile_features(localscribe_llama PRIVATE cxx_std_17)
target_link_libraries(localscribe_llama PRIVATE llama)
if(LOCALSCRIBE_AVX_VARIANT)
    set_target_properties(localscribe_llama PROPERTIES OUTPUT_NAME localscribe_llama_avx)
    target_compile_definitions(localscribe_llama PRIVATE LOCALSCRIBE_AVX_VARIANT)
endif()
if(ANDROID)
    target_link_libraries(localscribe_llama PRIVATE log)
    target_link_options(localscribe_llama PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-Bsymbolic" "-Wl,--exclude-libs,ALL")
else()
    find_package(JNI REQUIRED)
    target_include_directories(localscribe_llama PRIVATE ${JNI_INCLUDE_DIRS})
endif()
